cmake_minimum_required(VERSION 3.29)

set(CMAKE_XCODE_GENERATE_TOP_LEVEL_PROJECT_ONLY TRUE)
project(AppleNativeWidgets LANGUAGES Swift)

set(CMAKE_OSX_DEPLOYMENT_TARGET "14.0" CACHE STRING "" FORCE)
set(WIDGET_ROOT "${CMAKE_CURRENT_LIST_DIR}/..")

add_executable(BJTUHPCWidgetExtension MACOSX_BUNDLE
  "${WIDGET_ROOT}/HPC/HPCWidget.swift"
)
set_target_properties(BJTUHPCWidgetExtension PROPERTIES
  BUNDLE_EXTENSION "appex"
  MACOSX_BUNDLE_INFO_PLIST "${WIDGET_ROOT}/HPC/ExtensionInfo.plist"
  XCODE_PRODUCT_TYPE "com.apple.product-type.app-extension"
  XCODE_EXPLICIT_FILE_TYPE "wrapper.app-extension"
  XCODE_ATTRIBUTE_PRODUCT_BUNDLE_IDENTIFIER "com.example.bjtu-hpc-native-widget.widget"
  XCODE_ATTRIBUTE_CODE_SIGN_ENTITLEMENTS "${WIDGET_ROOT}/Widget.entitlements"
  XCODE_ATTRIBUTE_CODE_SIGN_STYLE "Manual"
  XCODE_ATTRIBUTE_CODE_SIGN_IDENTITY "-"
  XCODE_ATTRIBUTE_SWIFT_ACTIVE_COMPILATION_CONDITIONS "WIDGET_EXTENSION"
  XCODE_ATTRIBUTE_SWIFT_VERSION "6.0"
  XCODE_ATTRIBUTE_APPLICATION_EXTENSION_API_ONLY "YES"
  XCODE_ATTRIBUTE_SKIP_INSTALL "NO"
)

add_executable(BJTUHPCNativeWidget MACOSX_BUNDLE
  "${WIDGET_ROOT}/HPC/HPCApp.swift"
)
set_target_properties(BJTUHPCNativeWidget PROPERTIES
  MACOSX_BUNDLE_INFO_PLIST "${WIDGET_ROOT}/HPC/AppInfo.plist"
  XCODE_ATTRIBUTE_PRODUCT_BUNDLE_IDENTIFIER "com.example.bjtu-hpc-native-widget"
  XCODE_ATTRIBUTE_CODE_SIGN_STYLE "Manual"
  XCODE_ATTRIBUTE_CODE_SIGN_IDENTITY "-"
  XCODE_ATTRIBUTE_SWIFT_VERSION "6.0"
  XCODE_ATTRIBUTE_SKIP_INSTALL "NO"
  XCODE_EMBED_APP_EXTENSIONS "BJTUHPCWidgetExtension"
  XCODE_EMBED_APP_EXTENSIONS_CODE_SIGN_ON_COPY "YES"
)
